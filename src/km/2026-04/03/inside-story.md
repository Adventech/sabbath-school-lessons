### ទាល់តែក្លាយជាអ្នកដឹកនាំ អាត់វេនទីស្ទ

(ដោយ និពន្ធនាយកផ្នែកបេសកកម្ម)

ជីវិតរបស់ ម៉ារ៉េក បានចាប់ផ្ដើមផ្លាស់ប្ដូរ ខណៈពេលដែលគាត់កំពុងតែអានសៀវភៅ មហាវិវាទ នៅក្នុងប្រទេសប៉ូឡូញ។ គាត់បានសម្រេចចិត្តរក្សាថ្ងៃបរិសុទ្ធ ដែលជាថ្ងៃទីប្រាំពីរ។ គាត់ចង់ផ្ដាច់បារីណាស់ ប៉ុន្តែមិនអាចធ្វើទៅបានឡើយ។

មានថ្ងៃសៅរ៍មួយ គាត់បានដើរកាត់ផ្ទាំងប៉ាណូផ្សព្វផ្សាយមួយនៅតាមផ្លូវ ដែលបានផ្ដល់នូវវគ្គសិក្សាផ្ដាច់បារី រយៈពេលប្រាំថ្ងៃ នៅឯក្រុមជំនុំ អាត់វេនទីស្ទ មួយកន្លែង។ ម៉ារ៉េក មិនដែលបានឮអំពីឈ្មោះក្រុមជំនុំនិកាយនេះសោះឡើយ សូម្បីតែបន្ទាប់ពីបានអានសៀវភៅ មហាវិវាទ និងបានទៅលេងហាងលក់សៀវភៅ អាត់វេនទីស្ទ ដែលជាកន្លែង ដែលគាត់បានទិញសៀវភៅនោះរួចហើយក៏ដោយ។

ម៉ារ៉េក បានដើរសំដៅត្រង់ទៅទីនោះភ្លាម។ ពេលនោះគឺជាម៉ោងពីររសៀល ថ្ងៃបរិសុទ្ធ ទៅហើយ ដែលជាធម្មតាព្រះវិហារ មុខជាត្រូវនៅទំនេរ បន្ទាប់ពីកម្មវិធីថ្វាយបង្គំរួច ប៉ុន្តែមានក្រុមអ្នកផ្សព្វផ្សាយដំណឹងល្អខាងឯកសារបោះពុម្ពមួយក្រុម កំពុងស្ថិតនៅទីនោះ ហើយពួកគេបានអញ្ជើញ ម៉ារ៉េក ឲ្យត្រឡប់មកវិញម្ដងទៀត នៅព្រឹកថ្ងៃបរិសុទ្ធសប្ដាហ៍ក្រោយ។

ម៉ារ៉េក បានត្រឡប់មកវិញនៅថ្ងៃបរិសុទ្ធក្រោយ ហើយបានរីករាយស្ដាប់សេចក្ដីអធិប្បាយ។ គាត់មានការកោតសរសើរជាខ្លាំង ចំពោះសេចក្ដីសប្បុរសរបស់សមាជិកក្រុមជំនុំ។ ពួកគេបានប្រាប់គាត់ថា មានក្រុមយុវជនមួយក្រុមធំ នឹងមកជួបជុំគ្នានៅឯទីក្រុងមួយ ដែលនៅក្បែរនោះ នៅថ្ងៃបរិសុទ្ធ ក្រោយ ហើយបានបបួលគាត់ឲ្យទៅជាមួយពួកគេដែរ។

នៅព្រឹកថ្ងៃបរិសុទ្ធបន្ទាប់ ខណៈពេលដែល ម៉ារ៉េក កំពុងតែរង់ចាំពួកអ្នក អាត់វេនទីស្ទ មកទទួលរូបគាត់ គាត់បានដឹងថា មានសំឡេងពីរកំពុងតែនិយាយមកកាន់គាត់។ សំឡេងមួយបានប្រាប់ថា៖ “ចូរនៅខាងហ្នឹងហើយ ហើយរីករាយនឹងអាកាសធាតុដ៏ល្អនេះចុះ”។ ឯសំឡេងមួយទៀតបានដាស់តឿនថា៖ “ចូរចាំនៅទីនេះចុះ ពីព្រោះវាពិតជាមានសារៈសំខាន់ណាស់ ដែលលោកត្រូវទៅចូលរួមការប្រជុំនេះ”។ ពួកអ្នក អាត់វេនទីស្ទ បានមកដល់ ហើយបាននាំគាត់ទៅកាន់កិច្ចប្រជុំនោះ។ កម្មវិធីថ្វាយបង្គំបានធ្វើឲ្យ ម៉ារ៉េក មានការកោតសរសើរជាខ្លាំង។ គ្រប់ពាក្យសម្ដីទាំងអស់សុទ្ធតែមានន័យសម្រាប់រូបគាត់។ អ្នកប្រកាសដំណឹងល្អ ដែលជាគ្រូគង្វាល ជនជាតិអង់គ្លេសម្នាក់ មកពីទីក្រុងឡុងដ៍ បានអធិប្បាយរហូតដល់ម៉ោងដប់ពីរថ្ងៃត្រង់។ បន្ទាប់មក គាត់មានប្រសាសន៍ថា៖ “ខ្ញុំដឹងថា ខ្ញុំគួរតែបញ្ចប់នៅត្រឹមនេះហើយ ប៉ុន្តែខ្ញុំដឹងថា ច្បាស់ជាមាននរណាម្នាក់នៅទីនេះដែលកំពុងតែត្រូវការព្រះយេស៊ូវ”។

ម៉ារ៉េក បានគិតថា តើនរណាទៅដែលបានប្រាប់គាត់អំពីរឿងរបស់ខ្ញុំនោះ? បន្ទាប់មក គ្រូគង្វាលរូបនោះ បានចែករំលែករឿងរ៉ាវផ្ទាល់ខ្លួនរបស់គាត់។ គាត់បានកើតមកក្នុងគ្រួសារដែលកាន់សាសនា ប៉ុន្តែបានចាកចេញពីក្រុមជំនុំ។ គាត់ធ្លាប់បានប្រើប្រាស់គ្រឿងញៀន និងផឹកស្រា។ ក្រុមជំនុំរបស់គាត់ និងសូម្បីតែម្ដាយបង្កើតរបស់គាត់ ក៏បានឈប់អធិស្ឋានសម្រាប់គាត់ទៅហើយដែរ។

គ្រូគង្វាលមានប្រសាសន៍ថា៖ “បន្ទាប់មក ខ្ញុំបានជួបនឹងព្រះយេស៊ូវ។ ព្រះអង្គបានលើកខ្ញុំឡើងពីបាតជម្រៅ ដែលខ្ញុំធ្លាប់រស់នៅ ហើយឥឡូវនេះ ខ្ញុំនៅទីនេះដើម្បីប្រាប់លោកអ្នករាល់គ្នាអំពីព្រះចេស្តា របស់ព្រះអង្គ។ ហើយទ្រង់អាចផ្លាស់ប្ដូរជីវិតរបស់លោកអ្នកបាន។ ប្រសិនបើលោកអ្នកចង់ឲ្យព្រះយេស៊ូវផ្លាស់ប្ដូរជីវិតរបស់លោកអ្នក សូមគ្រាន់តែដើរមកខាងមុខនេះចុះ”។ ម៉ារ៉េក បានក្រឡេកមើលឃើញមនុស្សចំនួន ១,០០០ នាក់ នៅក្នុងចំណោមហ្វូងមនុស្ស ហើយមានចិត្តរារែកអន្ទះសា។ ប៉ុន្តែ នៅក្នុងប្រយោគបន្ទាប់ភ្លាម គ្រូគង្វាល ក៏បានដោះស្រាយនូវសេចក្ដីសង្ស័យរបស់គាត់ចំៗតែម្ដង។

គ្រូគង្វាលមានប្រសាសន៍ទៀតថា៖ “កុំគិតអំពីអ្នកដទៃដែលកំពុងតែសម្លឹងមើលលោកអ្នកអី។ សូមគ្រាន់តែដើរមកខាងមុខចុះ។ នេះជារឿងរវាងលោកអ្នក និងព្រះតែប៉ុណ្ណោះ”។

ម៉ារ៉េក បានក្រោកឈរឡើង។ គាត់មិនអាចអង្គុយបានទៀតឡើយ។ បេះដូងរបស់គាត់បានលោតញាប់យ៉ាងខ្លាំង ខណៈពេលដែលគាត់ដើរទៅខាងមុខ។ នៅពេលដែលគ្រូគង្វាល បានអធិស្ឋាន ម៉ារ៉េក បានយល់ដឹងអំពីផែនការសង្គ្រោះជាលើកដំបូងបំផុត នៅក្នុងជីវិតរបស់គាត់។ គាត់បានចាប់ផ្ដើមយំ ហើយគ្មានអ្វីអាចឃាត់គាត់បានឡើយ។ ទឹកភ្នែកបានហូរមកដោយទប់មិនបាន។ ប៉ុន្តែ ទោះជាយ៉ាងណា ម៉ារ៉េក ក៏មានចិត្តរំភើប និងមានសេចក្តីសុខសាន្ត ព្រមទាំងសេចក្ដីត្រេកអរជាខ្លាំងដែរ។

នៅសព្វថ្ងៃនេះ ម៉ារ៉េក គ្មានសេចក្ដីស្ដាយក្រោយទាល់តែសោះ។ គាត់កំពុងតែបម្រើការជាអ្នកដឹកនាំក្រុមជំនុំម្នាក់ នៅក្នុងប្រទេសប៉ូឡូញ ហើយនៅតែបន្តចែករំលែកជំនឿរបស់គាត់ ទៅកាន់អ្នកដទៃជានិច្ច៕
