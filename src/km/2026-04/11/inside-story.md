### ពួកអារក្ស និងសេចក្ដីស្លាប់៖ ភាគទី១

ដោយ៖ និពន្ធនាយកផ្នែកបេសកកម្ម

ក្នុងអំឡុងពេលនៃការសិក្សាព្រះគម្ពីររួមគ្នា នៅក្នុងបន្ទប់ផ្ទះល្វែងរបស់នាង នាទីក្រុង រ៉ាហេ (Raahe) ដែលជាក្រុងមួយរបស់ប្រទេសហ្វាំងឡង់ ស្ថិតនៅចម្ងាយប្រហែល ៦០០គីឡូម៉ែត្រ ភាគខាងជើងនៃទីក្រុងហ្វារ ហែលស៊ីនគី (Helsinki) នោះ កញ្ញា លីណា បានងាកទៅរកមិត្តភក្តិរបស់នាង គឺអានណេលី និងលោក ទីម៉ូ។

នាងនិយាយថា៖ “ខ្ញុំមានអារម្មណ៍មិនល្អទាល់តែសោះ។ តោះ យើងអធិស្ឋាន”។ និស្សិតសាកលវិទ្យាល័យទាំងបីនាក់ ក៏បានលុតជង្គង់ចុះ។ នៅក្នុងខណៈពេលនោះ ស្រាប់តែមានស្រមោលខ្មៅដ៏ខ្ពស់មួយ បានលេចចូលមកក្នុងបន្ទប់ទទួលភ្ញៀវ រួចហើយក៏ស្ទុះដេញសំដៅទៅរកកញ្ញា អានណេលី។ នាងមានអារម្មណ៍រន្ធត់ចិត្តជាខ្លាំង ខណៈពេលដែលស្រមោលខ្មៅនោះ ព្យាយាមលូកចាប់យករូបនាង។ ចំណែកឯលោក ទីម៉ូ និងនាង លីណា វិញ បានអធិស្ឋានកាន់តែខ្លាំងក្លាឡើងដោយអស់ពីចិត្ត។

បន្ទាប់មក មានស្រមោលមានពន្លឺមួយ បានលេចមកក្នុងបន្ទប់ ហើយបានបណ្ដេញស្រមោលខ្មៅនោះឲ្យចេញទៅ។ ស្រមោលខ្មៅនោះ បានឈរនៅក្បែរទ្វារ ហើយព្យាយាមចូលមកខាងក្នុងបន្ទប់វិញ ប៉ុន្តែស្រមោលមានពន្លឺ បានរារាំងរាល់គ្រប់ការប៉ុនប៉ងទាំងអស់។ បន្ទាប់ពីការតស៊ូអស់រយៈពេលប្រហែល១០នាទី ស្រមោលខ្មៅនោះ ក៏បានចុះចាញ់ ហើយចាកចេញបាត់ទៅ ។

នៅពេលដែលភាពស្ងប់ស្ងាត់បានត្រឡប់មកក្នុងបន្ទប់វិញ និស្សិតដែលកំពុងតែមានការភ័យញាប់ញ័រទាំងនោះ បានចាប់ផ្ដើមនិយាយរៀបរាប់ដំណើររឿងរ៉ាវឡើងវិញ អំពីអ្វីដែលបានកើតឡើង។ លីណា បានរៀបរាប់ពីការវាយប្រហារគ្នាទៅវិញទៅមក រវាងស្រមោលមានពន្លឺ និងស្រមោលខ្មៅ។ ទីម៉ូ មើលឃើញស្រមោលមានពន្លឺ និងស្រមោលខ្មៅ ដេញឆ្វៀលឆ្ងាត់កាត់មុខគាត់នៅលើឥដ្ឋការ៉ូក្នុងបន្ទប់តែប៉ុណ្ណោះ។ ចំណែកឯនាង អានណេលី វិញ គឺមិនចង់និយាយពីអ្វី ដែលនាងបានមើលឃើញនោះឡើយ។

លុះក្រោយមក ពួកគេបានដឹងព័ត៌មានថា មានការវាយប្រហាររបស់វិញ្ញាណអាក្រក់មួយ បានកើតឡើងនៅចំពេលវេលាដែលមានព្រឹត្តិការណ៍អត្តឃាត នៅក្នុងផ្ទះមួយដែលស្ថិតនៅក្បែរខាង។ កញ្ញា លីណា បាននិយាយថា៖ “នោះហើយជាមូលហេតុ ដែលខ្ញុំមានអារម្មណ៍មិនល្អបែបនោះ”។

ដោយសម្ងំស្ងៀមតទៅទៀតមិនបាន កញ្ញា អានណេលី បានទទួលស្គាល់ថា ខ្លួននាងធ្លាប់បានប្រើមន្តអាគមកាលពីមុន ហើយនាងនៅតែបន្តរងការរំខានធ្វើបាបពីពួកវិញ្ញាណអាក្រក់នៅឡើយ។ ទោះជាយ៉ាងណាក៏ដោយ នាងនិយាយថា ព្រះទ្រង់គឺមានអំណាចខ្លាំងក្លាជាងពួកវាឆ្ងាយណាស់។ ក្នុងពេលដែល នៅលើគ្រែតែម្នាក់ឯង បន្ទាប់ពីការវាយប្រហារនោះមក នាងបានមើលឃើញស្រមោលមានពន្លឺនោះ ដើរចូលមកក្នុងបន្ទប់របស់នាង ហើយអង្គុយនៅលើគ្រែជាមួយនាង រហូតដល់មេឃភ្លឺស្រាងៗ។

ការវាយប្រហារពីពួកអារក្សបានបញ្ឈប់ទៅវិញ បន្ទាប់ពីកញ្ញា អានណេលី បានទទួលបុណ្យជ្រមុជទឹកចូលជាសមាជិក នៃក្រុមជំនុំសេវេនដេយ អាត់វេនទីស្ទ។ នេះគឺជាលើកដំបូងបង្អស់ ដែលលោក ទីម៉ូ បានជួបប្រទះ និងបានឃើញផ្ទាល់ភ្នែកនូវមហាវិវាទរវាងព្រះគ្រីស្ទ និងសាតាំងបែបនេះ ហើយវាមិនមែនជាលើកចុងក្រោយនោះឡើយ។ លោក ទីម៉ូ ដែលតាមធម្មតាជាមនុស្សគេងលក់ស្កប់ស្កល់ណាស់ ស្រាប់តែនៅយប់មួយ លោកបានភ្ញាក់ដឹងខ្លួនឡើង ទាំងមានអារម្មណ៍ដឹងថា មាននរណាម្នាក់កំពុងតែឈរសម្លឹងមើលលោក នៅក្នុងភាពងងឹត។ បន្ទាប់មក លោកក៏បានឮសំឡេងមួយ។ សំឡេងនោះនិយាយថា៖ “កុំទទួលបុណ្យជ្រមុជទឹកឲ្យសោះ”។ លោក ទីម៉ូ ដែលជានិស្សិតផ្នែកវិស្វកម្មកម្មវិធីកុំព្យូទ័រ កំពុងតែរៀបចំខ្លួន ដើម្បីទទួលបុណ្យជ្រមុជទឹកចូលក្នុងក្រុមជំនុំអាត់វេនទីស្ទ។ លោកបានសម្លឹងមើលទៅក្នុងភាពងងឹត។ លោកមើលឃើញអ្វីក៏មិនឃើញដែរ ប៉ុន្តែ លោកអាចមានអារម្មណ៍ដឹងពីវត្តមានរបស់ម្នាក់នោះ។ លោកក៏បានអធិស្ឋាន ហើយវត្តមាននោះក៏បានចាកចេញបាត់ទៅ។

នៅថ្ងៃបន្ទាប់ កញ្ញា លីណា បានប្រាប់លោក ទីម៉ូ ថា មានមនុស្សម្នាក់បានធ្វើអត្តឃាតសម្លាប់ខ្លួន នៅក្បែរផ្ទះរបស់លោក កាលពីយប់មិញ។

លោក ទីម៉ូ បានសួរថា៖ “តើអ្នកដឹងទេថា រឿងនោះបានកើតឡើងនៅម៉ោងប៉ុន្មានដែរ?” នាងថា នាងដឹង វាជាពេលដែលលោកបានភ្ញាក់ដឹងខ្លួនពីដំណេកនោះឯង៕
