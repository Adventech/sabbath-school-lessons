### ពីក្ដីអស់សង្ឃឹម ទៅកាន់វាសនាដែលព្រះបានចាត់តាំង

នៅក្នុងគ្រាដ៏ត្រជាក់ស្រេបនៃខែមករា ឆ្នាំ១៩០៦ យុវជនជនជាតិស្លូវេនីម្នាក់ ឈ្មោះថា អាល់ប៊ីន ម៉ូឈ្នីគ (Albin Močnik) បានឈរនៅលើស្ពានមួយ ក្នុងក្រុង ប្រេមឺហាវិន (Bremerhaven) នៃប្រទេសអាល្លឺម៉ង់ ទាំងក្នុងចិត្តកំពុងតែគិតចង់សម្លាប់ខ្លួន។ ប៉ុន្តែ នៅខណៈពេលដែលលោកកំពុងតែស្ថិតនៅលើគែម នៃក្ដីអស់សង្ឃឹមខ្លាំងបំផុតនោះ មានជនប្លែកមុខម្នាក់ បានចូលមកជួយអន្តរាគមន៍ ដាស់តឿនតម្រង់ផ្លូវលោក ឱ្យទៅជួបនឹងគ្រួសារអាត់វេនទីស្ទ ដែលជាងធ្វើស្បែកជើង ដែលពួកគេបានជួយផ្លាស់ប្ដូរដំណើរប្រវត្តិសាស្ត្រសម្រាប់តំបន់បាល់កង់ (Balkans) ដោយមិនដឹងខ្លួន។

លោក អាល់ប៊ីន បានផ្លាស់ទៅរស់នៅជាមួយគ្រួសារ ស៊ីហ្វឺរ (Schiefers) និងគ្រួសារ បាណាស (Banas)។ សេចក្ដីបន្ទាល់ដ៏ស្ងប់ស្ងាត់តាមរយៈការរស់នៅរបស់ពួកគេ បានដាស់ចិត្តរបស់លោក ហើយនៅត្រឹមខែមីនា ឆ្នាំ១៩០៦ លោកក៏បានទទួលបុណ្យជ្រមុជទឹក។ នេះបានគូសបញ្ជាក់ពីការចាប់ផ្ដើម នៃបេសកកម្មពេញមួយជីវិត ដែលនឹងជួយស្ថាបនាក្រុមជំនុំ អាត់វេនទីស្ទ នៅក្នុងអតីតប្រទេសយូហ្គោស្លាវី (Yugoslavia)។

មិនយូរប៉ុន្មាន បន្ទាប់ពីបានទទួលបុណ្យជ្រមុជទឹករួចមក លោក អាល់ប៊ីន បានចូលរួមមហាសន្និបាតសហភាពអាល្លឺម៉ង់ ដែលត្រូវបានរៀបចំឡើងនៅក្នុងក្រុង ហ្វ្រីដិនសូ (Friedensau) ដោយមិនបានដឹងខ្លួនទាល់តែសោះថា លោកគឺជាអ្នកអាត់វេនទីស្ទ ជនជាតិស្លូវេនី ដំបូងគេបង្អស់។ ដោយទទួលបានការលើកទឹកចិត្តពីលោក ចន អេហ្វ ហ៊ូនឺកាត (John F. Huenergardt) ឱ្យទៅផ្សាយដំណឹងល្អដល់ជនជាតិរបស់លោក នោះលោកក៏បានចុះឈ្មោះចូលរៀននៅសាលា ហ្វ្រីដិនសូ ជាកន្លែងដែលលោកមានចិត្តស្រឡាញ់ខ្លាំង នោះបេសកកម្មរបស់លោក បានចម្រើនលូតលាស់ឡើង។ លោកបានចាប់ផ្ដើមបកប្រែ និងចែកចាយមេរៀនថ្នាក់ថ្ងៃបរិសុទ្ធទៅជាភាសា ស៊ែបបូ-ក្រូអាត (Serbo-Croatian)។

នៅក្នុងឆ្នាំ១៩០៨ ក្ដីស្រមៃរបស់លោក អាល់ប៊ីន បានលេចចេញជារូបរាង។ នៅពេលបញ្ចប់ការសិក្សារបស់ខ្លួន លោកបានធ្វើដំណើរទៅកាន់ប្រទេសក្រូអាស៊ី ជាមួយគ្រូគង្វាលម្នាក់ឈ្មោះ រ៉ូបឺត ហ្ស៊ីលីងហ្គឺរ (Robert Schillinger)។ ទោះបីជាមានឧបសគ្គខាងភាសា និងឧបសគ្គខាងផ្លូវច្បាប់ក៏ដោយ ក៏លោក អាល់ប៊ីន បានចែករំលែកដំណឹងល្អនៅក្នុងក្រុងហ្សាគ្រីប (Zagreb) និងក្រុងស្ព្លីត (Split) ដែរ។ លោកបានផ្លាស់ទៅរស់នៅក្នុងតំបន់ស្រេម (Srem) ដែលកាលនោះជាផ្នែកមួយនៃប្រទេសក្រូអាស៊ី ហើយបានផ្សងព្រេងចូលទៅក្នុងប្រទេសស៊ែប៊ី ទាំងប្រកាសដំណឹងល្អ និងធ្វើបុណ្យជ្រមុជទឹកដល់ពួកអ្នកជឿថ្មីៗ ដែលជាញឹកញាប់ ត្រូវលះបង់ដោយការចាប់គុកច្រវាក់។

ទោះបីជាលោកមានជំងឺមកបៀតបៀនម្ដងហើយម្ដងទៀត និងមានការប្រឆាំងខាងនយោបាយក៏ដោយ ក៏លោកអាល់ប៊ីន មិនដែលបោះបង់ចោលបេសកកម្មរបស់លោកឡើយ។ នៅក្នុងអំឡុងសង្គ្រាមលោកលើកទី១ លោកបានទ្រាំទ្រនឹងកាតព្វកិច្ចយោធា ស្ថានភាពសុខភាពដ៏ធ្ងន់ធ្ងរ និងក្ដីទុក្ខសោកនៃការបែកបាក់ពីគ្រួសារ ដែលកំពុងតែមានសមាជិកកើនឡើង។ ប៉ុន្តែ ទោះជាយ៉ាងណា សូម្បីតែនៅក្នុងអំឡុងពេលសម្រាកព្យាបាលជំងឺក៏ដោយ ក៏លោកបានតែងសៀវភៅបទទំនុកតម្កើង អាត់វេនទីស្ទ ដំបូងគេបង្អស់ ជាភាសាស៊ែបបូ-ក្រូអាត ដោយលោកផ្ទាល់បាននិពន្ធបទចម្រៀងចំនួន១២០បទ។

ក្រោយពីសង្គ្រាមបានបញ្ចប់ លោក អាល់ប៊ីន បានដឹកនាំការខិតខំប្រឹងប្រែង ដើម្បីប្រឆាំងនឹងចលនាកែទម្រង់ ដែលបង្កឱ្យមានការបែកបាក់ និងបានខិតខំកសាងក្រុមជំនុំអាត់វេនទីស្ទឡើងវិញ។ ចលនានេះបានចាប់ផ្ដើមឡើងនៅមុនសង្គ្រាមលោកលើកទី១ នៅពេលដែលមានការយល់ឃើញខុសគ្នា ស្តីអំពីកាតព្វកិច្ចយោធាបានកើតឡើងក្នុងចំណោមគ្រីស្ទានអាត់វេនទីស្ទ ជនជាតិអាល្លឺម៉ង់។ នៅទីបំផុត កិច្ចការបម្រើរបស់លោក អាល់ប៊ីន បានបង្កើតជាផលផ្លែផ្កា៖ លោកបានប្រោសឱ្យក្រុមជំនុំមានភាពរស់រវើកឡើងវិញ បានបណ្ដុះបណ្ដាលអ្នកដឹកនាំថ្មីៗ ហើយនៅត្រឹមឆ្នាំ១៩២៥ លោកត្រូវបានគេបោះឆ្នោតជ្រើសតាំងជាប្រធានដំបូងគេបង្អស់ នៃមហាសន្និបាតសហភាពយូហ្គោស្លាវី (Yugoslav Union Conference)។

ក្នុងនាមជាប្រធាន លោក អាល់ប៊ីន ម៉ូឈ្នីគ បានផ្ដល់ជាសំឡេងជាសាធារណៈ ដល់ជំនឿ អាត់វេនទីស្ទ នៅក្នុងស្ថានភាពនយោបាយដ៏មានអរិភាព (ភាពជាសត្រូវ)។ សៀវភៅរបស់លោកដែលមានចំណងជើងថា “គោលលទ្ធិអាត់វេនទីស្ទ” បោះពុម្ពក្នុងឆ្នាំ១៩២៥ បានជួយបកស្រាយពន្យល់យ៉ាងច្បាស់លាស់ ពីជំនឿរបស់ក្រុមជំនុំជូនដល់រដ្ឋាភិបាល ហើយបានជួយឱ្យនិកាយនេះ ទទួលបានការទទួលស្គាល់ស្របច្បាប់ជាផ្លូវការ។ លោកបានដឹកនាំកិច្ចការបកប្រែ និងធ្វើជានិពន្ធនាយកកាសែតរបស់ក្រុមជំនុំ គឺកាសែត ហ្គ្លាសនីក (Glasnik)។ លោកក៏បានបង្កើតសាលារៀន សម្រាប់បណ្ដុះបណ្ដាលពួកគ្រូគង្វាលនៅក្នុងឆ្នាំ១៩៣១ ផងដែរ។

កេរដំណែលរបស់លោក អាល់ប៊ីន ម៉ូឈ្នីគ ត្រូវបានឆ្លាក់ទុក តាមរយៈសេចក្ដីទុក្ខលំបាក និងជំនឿដ៏ឥតងាករេ។ ចាប់តាំងពីខណៈពេលដែល លោកបានជ្រើសរើសយកជីវិតនៅលើស្ពាន ក្នុងក្រុងប្រេមឺហាវិន រហូតដល់សេចក្ដីបន្ទាល់ដ៏ក្លាហានរបស់លោក នៅទូទាំងប្រទេសក្រូអាស៊ី និងតំបន់ឆ្ងាយៗដទៃទៀត នោះលោក អាល់ប៊ីន ម៉ូឈ្នីគ បានក្លាយជាអ្នកត្រួសត្រាយផ្លូវនៃសេចក្ដីសង្ឃឹម។ រឿងរ៉ាវរបស់លោក គឺជាសេចក្ដីបន្ទាល់មួយ អំពីអំណាចនៃការផ្លាស់ប្ដូរជីវិតនៃដំណឹងល្អ និងភាពក្លាហានរបស់អស់អ្នកដែលនាំយកដំណឹងល្អនោះ ទៅកាន់ព្រំដែនថ្មីៗ៕
