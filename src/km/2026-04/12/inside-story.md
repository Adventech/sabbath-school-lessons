### ពួកអារក្ស និងសេចក្ដីស្លាប់៖ ភាគទី២

ដោយ៖ និពន្ធនាយកផ្នែកបេសកកម្ម

សេចក្ដីសង្ខេបរឿងកន្លងមក៖ លោក ទីម៉ូ និងមិត្តភក្តិរបស់លោក គឺកញ្ញា លីណា និងកញ្ញា អានណេលី បានជួបប្រទះនឹងការប្រឈមមុខដ៏គួរឲ្យភ័យខ្លាច ជាមួយនឹងរឿងរ៉ាវអស្ចារ្យបែបអធិធម្មជាតិ ក្នុងអំឡុងពេលសិក្សាព្រះគម្ពីររួមគ្នា នៅក្នុងបន្ទប់ផ្ទះល្វែងរបស់កញ្ញា លីណា។ លុះក្រោយមក កញ្ញា អានណេលី បានបកស្រាយពន្យល់ថា នាងធ្លាប់បានប្រើមន្តអាគម កាលពីមុន ហើយពេលខ្លះ នាងនៅតែបន្តរងការរំខានពីពួកវិញ្ញាណអាក្រក់នៅឡើយ។ ទោះជាយ៉ាងណាក៏ដោយ នៅពេលដែលនាងបានទទួលបុណ្យជ្រមុជទឹកចូលជាសមាជិក នៃក្រុមជំនុំសេវេនដេយអាត់វេនទីស្ទ នោះការវាយប្រហារពីពួកអារក្សក៏បានបញ្ឈប់ទៅវិញ។ ចំណែកឯ លោក ទីម៉ូ វិញ គឺជានិស្សិតផ្នែកវិស្វកម្ម ដែលកំពុងតែរៀបចំខ្លួនទទួលបុណ្យជ្រមុជទឹកដូចគ្នាដែរ។ នៅយប់មួយ លោកបានភ្ញាក់ដឹងខ្លួនឡើង ដោយសារវត្តមានដ៏គួរឲ្យព្រឺព្រួចមួយ ដែលមកប្រាប់លោកថា៖ “កុំទទួលបុណ្យជ្រមុជទឹកឲ្យសោះ”។

ការព្រមាននាពេលរាត្រីនោះ គឺមិនអាចរារាំងលោក ទីម៉ូ មិនឲ្យទទួលបុណ្យជ្រមុជទឹកបានឡើយ ហើយលោកបានបន្តដំណើរទៅមុខ រហូតដល់បានក្លាយជាគ្រូគង្វាល អាត់វេនទីស្ទ ម្នាក់។ ប៉ុន្តែ ទោះជាយ៉ាងណា បទពិសោធន៍របស់លោក ជាមួយនឹងរឿងរ៉ាវបែបអធិធម្មជាតិ និងរឿងអត្តឃាត គឺមិនទាន់បានបញ្ចប់នៅឡើយទេ។

ថ្ងៃមួយ រូបលោក និងគ្រូគង្វាលអាត់វេនទីស្ទ មួយចំនួនទៀត បានឡើងជិះសាឡាងចម្លង ដើម្បីធ្វើដំណើរនៅពេលយប់ ទៅចូលរួមកម្មវិធីមហាសន្និបាតគ្រូគង្វាល ក្នុងប្រទេសស៊ុយអែត។ នៅយប់នោះ លោកមានអារម្មណ៍មិនស្ងប់ក្នុងចិត្ត។ បន្ទាប់ពីបានព្យាយាមដេក តែដេកមិនលក់សោះ គាត់ក៏មានអារម្មណ៍ថា ត្រូវតែក្រោកឡើង ហើយអធិស្ឋានជាបន្ទាន់។ នៅពេលដែលគាត់បានចាប់ផ្ដើមអធិស្ឋានភ្លាម គាត់ក៏បានឮសំឡេងសើចរបស់ពួកអារក្ស។ សំឡេងគួរឱ្យតក់ស្លុតនោះមិនអាចពិពណ៌នាបានឡើយ វាដូចជាសំឡេងសើចរបស់មនុស្សឆ្កួតលីលា។ លោក ទីម៉ូ មានអារម្មណ៍ថា វាហាក់ដូចជាមានរឿងអាក្រក់អ្វីមួយ កំពុងតែកើតឡើង ប៉ុន្តែគាត់មិនដឹងថាជារឿងអ្វីនោះទេ។ គាត់បានអធិស្ឋានអស់រយៈពេលពីរម៉ោង។

នៅពេលញ៉ាំអាហារពេលព្រឹក គ្រូគង្វាល មានវ័យចំណាស់ម្នាក់បានដើរចូលមកជិត ទីម៉ូ ហើយសួរថា៖ “តើយប់មិញ លោកមានរឿងអ្វីកើតឡើង? ពីព្រោះព្រះវិញ្ញាណបរិសុទ្ធបានប្រាប់ខ្ញុំឱ្យអធិស្ឋានឱ្យលោក”។ គាត់ភ្ញាក់ឡើងនៅពាក់កណ្តាលអធ្រាត្រ ហើយបានអធិស្ឋានឱ្យ ទីម៉ូ ។ បន្ទាប់មក គ្រូគង្វាលម្នាក់ទៀតក៏បានមកដល់តុអាហារពេលព្រឹក។

គាត់បាននិយាយថា៖ “អ្នករាល់គ្នាប្រហែលជាមិនដឹងថា មានរឿងអ្វីកើតឡើងចំពោះខ្ញុំកាលពីយប់មិញទេ។ ខ្ញុំភ្ញាក់ឡើង ហើយមានអារម្មណ៍ថា មានគេជំរុញយ៉ាងខ្លាំង ឱ្យត្រូវតែចេញទៅខាងក្រៅដើម្បីស្រូបយកខ្យល់អាកាសបរិសុទ្ធ។ នៅពេលខ្ញុំទៅដល់ជាន់ខាងលើនៃសាឡាង ខ្ញុំបានឃើញបុរសម្នាក់រៀបនឹងលោតចូលទៅក្នុងសមុទ្រ”។ គ្រូគង្វាលរូបនោះបានលួងលោមបុរសនោះ ឱ្យត្រឡប់ចូលមកក្នុងសាឡាងវិញ ហើយបន្ទាប់មក គាត់បានផ្តល់ប្រឹក្សាដល់បុរសនោះ អស់រយៈពេលមួយម៉ោង ដើម្បីកុំឱ្យគាត់សម្លាប់ខ្លួន។ នៅពេលដែលគ្រូគង្វាលទាំងបីរូបនោះ បានប្រៀបធៀបពេលវេលានៃបទពិសោធន៍របស់ពួកគេ នៅពេលយប់នោះ ពួកគេបានដឹងថា ហេតុការណ៍ទាំងបីបានកើតឡើងក្នុងពេលដំណាលគ្នាតែម្តង។

លោក ទីម៉ូ ដែលឥឡូវនេះមានអាយុ៤៥ឆ្នាំ ហើយជាប្រធាននាយកដ្ឋានទំនាក់ទំនង នៃព្រះវិហារអាត់-វេនទីស្ទ នៅប្រទេសហ្វាំងឡង់ លោកមើលឃើញថា ជំនួបទាំងបីលើកជាមួយនឹងការធ្វើអត្តឃាត និងអព្ភូត-ហេតុអរូបិយទាំងនេះ គឺជាភស្តុតាងដែលបង្ហាញថា មហាវិវាទរវាងព្រះគ្រីស្ទ និងសាតាំង គឺជាការប្រយុទ្ធដ៏ពិតប្រាកដ។

លោកបានមានប្រសាសន៍ថា៖ “វាកំពុងកើតឡើងនៅជុំវិញខ្លួនយើង។ អ្វីដែលជាការលើកទឹកចិត្តនោះគឺ ព្រះយេស៊ូវបានឈ្នះរួចស្រេចទៅហើយ។ យើងគ្មានអ្វីត្រូវភ័យខ្លាចឡើយ។ ទោះបីជាមានព្រឹត្តិការណ៍អព្ភូតហេតុ និងគួរឱ្យភ័យខ្លាចទាំងនេះក៏ដោយ ក៏ព្រះយេស៊ូវនៅតែការពារយើង។ គ្មានអ្វីដែលសាតាំងអាចធ្វើបានឡើយ”៕
